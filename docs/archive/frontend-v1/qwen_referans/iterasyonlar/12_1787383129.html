docker-compose.yml

hedef yapı
version: "3.9"
services:
  postgres:
    image: postgres:16-alpine
    volumes: [ "pg_data:/var/lib/postgresql/data" ]
  redis:
    image: redis:7-alpine
    command: redis-server --appendonly yes
  backend:
    build: ./backend          # FastAPI + event bus + plugin loader
    ports: [ "8000:8000" ]
    env_file: .env            # TESTNET API anahtari BURADA (sohbette paylasilmaz)
    depends_on: [ postgres, redis ]
    restart: unless-stopped
  frontend:
    build: ./frontend
    ports: [ "3000:80" ]
  backtest_worker:
    build: ./backend
    command: celery -A app.worker worker -c 4 -Q backtest
  bot_ai:
    build: ./backend
    command: python -m app.ai.copilot --memory vectordb/
    volumes: [ "ai_memory:/app/vectordb" ]
volumes: { pg_data: {}, ai_memory: {} }
