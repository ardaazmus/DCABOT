hedef yapı
version: "3.9"
services:
  postgres:
    image: postgres:16-alpine
    volumes: [ "pg_data:/var/lib/postgresql/data" ]
  redis:
    image: redis:7-alpine
    command: redis-server --appendonly yes
  backend:
    build: ./backend          # FastAPI + 5 engine
    ports: [ "8000:8000" ]
    depends_on: [ postgres, redis ]
    restart: unless-stopped
  frontend:
    build: ./frontend         # React + grafik motoru
    ports: [ "3000:80" ]
  backtest_worker:
    build: ./backend
    command: celery -A app.worker worker -c 4 -Q backtest
  regime_risk_engine:
    build: ./backend
    command: python -m app.engines.regime_risk
volumes: { pg_data: {} }
