version: "3.9"
services:
  postgres:
    image: postgres:16-alpine
    environment:
      POSTGRES_DB: hybridbot
      POSTGRES_PASSWORD_FILE: /run/secrets/pg_pass
    volumes: [ "pg_data:/var/lib/postgresql/data" ]
    healthcheck: { test: ["CMD-SHELL","pg_isready"], interval: 10s }
  redis:
    image: redis:7-alpine
    command: redis-server --appendonly yes
  backend:
    build: ./backend          # FastAPI + 5 engine
    ports: [ "8000:8000" ]
    env_file: .env
    depends_on: [ postgres, redis ]
    restart: unless-stopped
  frontend:
    build: ./frontend         # React + TV Lightweight Charts
    ports: [ "3000:80" ]
  backtest_worker:
    build: ./backend
    command: celery -A app.worker worker -c 4 -Q backtest
    depends_on: [ redis, postgres ]
  regime_risk_engine:
    build: ./backend
    command: python -m app.engines.regime_risk
    depends_on: [ redis ]
volumes: { pg_data: {} }
secrets:
  pg_pass: { file: ./secrets/pg_pass.txt }
